for col in df.select_dtypes(include="object").columns:
    df[col] = df[col].replace(["nan", "none", ""], np.nan)

    # Force consistent type
    df[col] = df[col].astype(str)

    # Convert back 'nan' string → real NaN
    df[col] = df[col].replace("nan", np.nan)

    # Safe mode extraction
    mode_val = df[col].mode()

    if len(mode_val) > 0:
        df[col].fillna(mode_val.iloc[0], inplace=True)
    else:
        df[col].fillna("Unknown", inplace=True)

def segment_customer(row):
    if (row['purchase_frequency'] == 'high') and (row['shopping_satisfaction'] >= 4):
        return "Frequent Buyers"
    
    elif (row['purchase_frequency'] == 'medium') and (row['shopping_satisfaction'] >= 3):
        return "Occasional Shoppers"
    
    elif (row['shopping_satisfaction'] <= 2):
        return "At-Risk Customers"
    
    else:
        return "Other"

df["customer_segment"] = df.apply(segment_customer, axis=1)
